// src/components/icons/IconWrapper.tsx

import { TypeSvgSvg } from '@type-dom/framework';

interface IconProps {
  svgClass: new () => TypeSvgSvg;
  size?: number;
  color?: string;
  className?: string;
}

export function IconWrapper({ 
  svgClass, 
  size = 24, 
  color,
  className 
}: IconProps) {
  const iconRef = useRef<TypeSvgSvg | null>(null);
  
  useEffect(() => {
    if (iconRef.current) {
      iconRef.current.resetSize(size, size);
      if (color) {
        iconRef.current.setStyle('fill', color);
      }
    }
  }, [size, color]);
  
  return <div ref={iconRef} className={className} />;
}

// 为所有图标添加 aria-label
interface AccessibleIconProps {
  svgClass: new () => TypeSvgSvg;
  label: string;
  decorative?: boolean;
}

function AccessibleIcon({ 
  svgClass: SvgClass, 
  label,
  decorative = false 
}: AccessibleIconProps) {
  const icon = new SvgClass();
  
  if (decorative) {
    icon.setAttr('aria-hidden', 'true');
  } else {
    icon.setAttr('role', 'img');
    icon.setAttr('aria-label', label);
  }
  
  return icon;
}

// 使用
