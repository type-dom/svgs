## 📋 测试体系总览

### 测试金字塔

```
        /\
       /  \
      / E2E \         端到端测试 (5%)
     /--------\       验证完整用户流程
    /          \
   / Integration \    集成测试 (15%)
  /----------------\   验证组件协作
 /                  \
/    Unit Tests      \  单元测试 (70%)
----------------------  验证独立功能单元
```

### 测试工具链

| 测试类型 | 工具 | 用途 | 覆盖率 |
|---------|------|------|--------|
| **单元测试** | Vitest | SVG 组件功能测试 | 70%+ |
| **集成测试** | Vitest + TypeDom | 组件集成验证 | 15%+ |
| **E2E 测试** | Playwright | 浏览器端完整流程 | 5%+ |
| **视觉回归** | Pixelmatch | SVG 渲染对比 | 关键组件 |
| **性能测试** | Vitest Benchmark | 渲染性能基准 | 核心组件 |

### 基本测试结构

```typescript
import { describe, it, expect, beforeEach } from 'vitest';
import { TdAddSvg } from '@type-dom/svgs';
import { TypeSvgSvg, SvgPath } from '@type-dom/framework';

describe('TdAddSvg', () => {
  let svg: TdAddSvg;
  
  beforeEach(() => {
    svg = new TdAddSvg();
  });
  
  describe('Initialization', () => {
    it('should create instance with default props', () => {
      expect(svg).toBeInstanceOf(TypeSvgSvg);
      expect(svg.className).toBe('TdAddSvg');
      expect(svg.viewBox).toBe('0 0 1024 1024');
    });
    
    it('should accept custom props', () => {
      const customSvg = new TdAddSvg({
        width: 32,
        height: 32,
        fill: 'red'
      });
      
      expect(customSvg.width).toBe(32);
      expect(customSvg.height).toBe(32);
    });
  });
  
  describe('Child Nodes', () => {
    it('should have correct number of paths', () => {
      expect(svg.childNodes).toBeInstanceOf(Array);
      expect(svg.childNodes.length).toBeGreaterThan(0);
    });
    
    it('should contain only SvgPath instances', () => {
      svg.childNodes.forEach(path => {
        expect(path).toBeInstanceOf(SvgPath);
      });
    });
    
    it('should have valid path data', () => {
      svg.childNodes.forEach(path => {
        expect(path.d).toBeDefined();
        expect(typeof path.d).toBe('string');
        expect(path.d.length).toBeGreaterThan(0);
      });
    });
  });
  
  describe('Size Management', () => {
    it('should reset size correctly', () => {
      svg.resetSize(48, 48);
      expect(svg.width).toBe(48);
      expect(svg.height).toBe(48);
    });
    
    it('should maintain aspect ratio', () => {
      svg.resetSize(100, 100);
      const ratio = svg.width / svg.height;
      expect(ratio).toBe(1);
    });
    
    it('should reject invalid dimensions', () => {
      expect(() => svg.resetSize(-1, 24)).not.toThrow();
      // 或者根据实现expect 抛出错误
    });
  });
  
  describe('DOM Operations', () => {
    it('should mount to container', () => {
      const container = document.createElement('div');
      svg.mount(container);
      
      expect(container.querySelector('svg')).toBeTruthy();
    });
    
    it('should add child nodes', () => {
      const initialCount = svg.childNodes.length;
      const newPath = new SvgPath({ d: 'M10 10 L20 20' });
      
      svg.addChild(newPath);
      expect(svg.childNodes.length).toBe(initialCount + 1);
    });
    
    it('should remove child nodes', () => {
      if (svg.childNodes.length > 1) {
        const toRemove = svg.childNodes[0];
        svg.removeChild(toRemove);
        expect(svg.childNodes).not.toContain(toRemove);
      }
    });
  });
  
  describe('Attribute Management', () => {
    it('should set attributes', () => {
      svg.setAttr('data-test', 'value');
      expect(svg.attrObj['data-test']).toBe('value');
    });
    
    it('should set styles', () => {
      svg.setStyle('opacity', '0.5');
      expect(svg.styleObj.opacity).toBe('0.5');
    });
    
    it('should update className', () => {
      expect(svg.className).toBe('TdAddSvg');
      svg.setAttr('class', 'custom-class');
      expect(svg.attrObj.class).toBe('custom-class');
    });
  });
});
```

## 🔗 集成测试规范 (Integration Testing)

### 组件集成测试

```typescript
import { describe, it, expect } from 'vitest';
import { CommonSvgs } from '@type-dom/svgs';
import { TypeDomApp } from '@type-dom/framework';

describe('SVG Components Integration', () => {
  let app: TypeDomApp;
  
  beforeEach(() => {
    app = new TypeDomApp({
      target: document.body
    });
  });
  
  afterEach(() => {
    app.destroy();
  });
  
  it('should render multiple SVGs together', () => {
    const { TdAddSvg, TdCloseSvg, TdUserSvg } = CommonSvgs;
    
    const container = document.createElement('div');
    const addSvg = new TdAddSvg();
    const closeSvg = new TdCloseSvg();
    const userSvg = new TdUserSvg();
    
    addSvg.mount(container);
    closeSvg.mount(container);
    userSvg.mount(container);
    
    const svgs = container.querySelectorAll('svg');
    expect(svgs.length).toBe(3);
  });
  
  it('should handle dynamic SVG switching', async () => {
    const { TdAddSvg, TdCloseSvg } = CommonSvgs;
    const container = document.createElement('div');
    
    let currentSvg = new TdAddSvg();
    currentSvg.mount(container);
    
    expect(container.querySelector('svg')).toBeTruthy();
    
    // Switch SVG
    currentSvg.unmount();
    currentSvg = new TdCloseSvg();
    currentSvg.mount(container);
    
    expect(container.querySelector('svg')).toBeTruthy();
  });
  
  it('should maintain consistent sizing across different icons', () => {
    const { TdAddSvg, ElArrowDownSvg, FlAccessTimeSvg } = CommonSvgs;
    
    const icons = [
      new TdAddSvg(),
      new ElArrowDownSvg(),
      new FlAccessTimeSvg()
    ];
    
    icons.forEach(icon => {
      icon.resetSize(24, 24);
      expect(icon.width).toBe(24);
      expect(icon.height).toBe(24);
    });
  });
});
```

### 与 TypeDom 框架集成

```typescript
import { describe, it, expect, beforeEach } from 'vitest';
import { TypeComponent, TypeElement } from '@type-dom/framework';
import { TdAddSvg } from '@type-dom/svgs';

class TestComponent extends TypeComponent {
  svg: TdAddSvg;
  
  constructor() {
    super();
    this.svg = new TdAddSvg();
  }
  
  render(): TypeElement {
    return this.svg as unknown as TypeElement;
  }
}

describe('TypeDom Integration', () => {
  it('should work within TypeComponent lifecycle', () => {
    const component = new TestComponent();
    const container = document.createElement('div');
    
    component.mount(container);
    expect(container.querySelector('svg')).toBeTruthy();
    
    component.unmount();
    expect(container.querySelector('svg')).toBeFalsy();
  });
  
  it('should handle reactive updates', async () => {
    const component = new TestComponent();
    const container = document.createElement('div');
    component.mount(container);
    
    // Update SVG properties
    component.svg.setAttr('fill', 'blue');
    
    // Wait for reactivity
    await new Promise(resolve => setTimeout(resolve, 0));
    
    const svgElement = container.querySelector('svg');
    expect(svgElement?.getAttribute('fill')).toBe('blue');
  });
});
```

### E2E 测试示例

```typescript
import { test, expect } from '@playwright/test';

test.describe('SVG Rendering E2E', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto('/');
  });
  
  test('should render SVG components correctly', async ({ page }) => {
    // Navigate to SVG showcase
    await page.click('[data-testid="svg-showcase"]');
    
    // Verify SVGs are rendered
    const svgs = page.locator('svg');
    await expect(svgs.first()).toBeVisible();
    
    // Check SVG attributes
    const firstSvg = svgs.first();
    await expect(firstSvg).toHaveAttribute('viewBox', '0 0 1024 1024');
    await expect(firstSvg).toHaveAttribute('class', /Td.*Svg/);
  });
  
  test('should handle SVG interactions', async ({ page }) => {
    await page.goto('/interactive');
    
    // Click to change SVG
    await page.click('[data-testid="change-svg-btn"]');
    
    // Verify SVG changed
    const newSvg = page.locator('svg[data-testid="new-icon"]');
    await expect(newSvg).toBeVisible();
  });
  
  test('should maintain performance with many SVGs', async ({ page }) => {
    await page.goto('/stress-test');
    
    // Wait for all SVGs to render
    await page.waitForSelector('svg', { state: 'attached' });
    
    const svgs = page.locator('svg');
    const count = await svgs.count();
    expect(count).toBeGreaterThan(100);
    
    // Check performance metrics
    const metrics = await page.evaluate(() => {
      return {
        fps: window.performance.now(),
        memory: (performance as any).memory?.usedJSHeapSize
      };
    });
    
    expect(metrics.fps).toBeDefined();
  });
  
  test('visual regression - SVG appearance', async ({ page }) => {
    await page.goto('/showcase');
    
    // Take screenshot and compare with baseline
    await expect(page).toHaveScreenshot('svg-showcase-baseline.png', {
      maxDiffPixels: 100,
      fullPage: false
    });
  });
});
```

## 🎯 Mock 与 Stub 规范

### Mock 数据模式

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { TdAddSvg } from '@type-dom/svgs';

// Mock TypeDom framework
vi.mock('@type-dom/framework', () => ({
  TypeSvgSvg: class MockTypeSvgSvg {
    className = '';
    childNodes = [];
    attrObj = {};
    styleObj = {};
    
    constructor(params: any = {}) {
      Object.assign(this, params);
    }
    
    resetSize = vi.fn();
    addChild = vi.fn();
    removeChild = vi.fn();
    mount = vi.fn();
    unmount = vi.fn();
  },
  
  SvgPath: class MockSvgPath {
    constructor(public params: any = {}) {}
  },
  
  addAttrObj: vi.fn()
}));

describe('Mock Testing Pattern', () => {
  it('should use mock for isolation', () => {
    const svg = new TdAddSvg();
    expect(svg.className).toBe('TdAddSvg');
  });
  
  it('should verify mock calls', () => {
    const svg = new TdAddSvg();
    svg.resetSize(24, 24);
    
    expect(svg.resetSize).toHaveBeenCalledWith(24, 24);
    expect(svg.resetSize).toHaveBeenCalledTimes(1);
  });
});
```
