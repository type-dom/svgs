# 单元测试指南

## 📋 概述

本指南详细说明了 TypeDOM SVG 组件库的单元测试编写规范、最佳实践和常见模式。

**最后更新**: 2026-03-18

## 🎯 单元测试原则

### FIRST 原则

| 原则                | 说明         | 示例             |
| ------------------- | ------------ | ---------------- |
| **Fast**            | 测试执行要快 | 单个测试 < 100ms |
| **Independent**     | 测试相互独立 | 无共享状态       |
| **Repeatable**      | 可重复运行   | 无随机失败       |
| **Self-validating** | 自动验证结果 | 清晰的断言       |
| **Timely**          | 及时编写     | TDD 或实现即测试 |

## 📁 测试文件结构

### 基本模板

```typescript
/**
 * @file TdAddSvg 组件测试
 * @description 测试 Add 图标的功能和属性
 */

import { describe, it, expect, beforeEach } from "vitest";
import { TdAddSvg } from "../../src/lib/common/add";

describe("TdAddSvg", () => {
  let svgInstance: TdAddSvg;

  beforeEach(() => {
    svgInstance = new TdAddSvg();
  });

  describe("Component Structure", () => {
    it("should have correct className", () => {
      expect(svgInstance.className).toBe("TdAddSvg");
    });

    it("should have childNodes array", () => {
      expect(Array.isArray(svgInstance.childNodes)).toBe(true);
    });

    it("should have viewBox attribute", () => {
      const viewBox = svgInstance.getAttribute("viewBox");
      expect(viewBox).toBe("0 0 1024 1024");
    });
  });

  describe("Size Management", () => {
    it("should have default size 24x24", () => {
      expect(svgInstance.props.width).toBe(24);
      expect(svgInstance.props.height).toBe(24);
    });

    it("should accept custom size props", () => {
      const customSvg = new TdAddSvg({ width: 32, height: 32 });
      expect(customSvg.props.width).toBe(32);
      expect(customSvg.props.height).toBe(32);
    });
  });

  describe("Path Data", () => {
    it("should have path data", () => {
      expect(svgInstance.childNodes.length).toBeGreaterThan(0);
    });

    it("should use currentColor for fill", () => {
      const path = svgInstance.childNodes[0];
      expect(path.attrObj.fill).toBe("currentColor");
    });
  });
});
```

### 3. Path 数据测试

#### 3.1 Path 存在性验证

```typescript
import { SvgPath } from "@type-dom/framework";
import { TdArrowSvg } from "../src/lib/common/arrow";

describe("Path Data Validation", () => {
  it("should contain at least one path", () => {
    const svg = new TdArrowSvg();
    expect(svg.childNodes.length).toBeGreaterThanOrEqual(1);
  });

  it("should have SvgPath instances", () => {
    const svg = new TdArrowSvg();
    svg.childNodes.forEach((child) => {
      expect(child instanceof SvgPath).toBe(true);
    });
  });

  it("should have valid path data", () => {
    const svg = new TdArrowSvg();
    const path = svg.childNodes[0] as SvgPath;
    const d = path.getData();

    expect(d).toBeDefined();
    expect(typeof d).toBe("string");
    expect(d.length).toBeGreaterThan(0);
  });
});
```

#### 3.2 Path 命令验证

```typescript
describe("Path Commands", () => {
  it("should start with Move command (M)", () => {
    const svg = new TdArrowSvg();
    const path = svg.childNodes[0] as SvgPath;
    const d = path.getData();

    expect(d).toMatch(/^M\s*\d+/);
  });

  it("should contain valid SVG commands", () => {
    const svg = new TdArrowSvg();
    const path = svg.childNodes[0] as SvgPath;
    const d = path.getData();

    // Common SVG path commands
    const validCommands = ["M", "L", "H", "V", "C", "S", "Q", "T", "A", "Z"];
    const commands = d.match(/[A-Z][a-z]*/g) || [];

    commands.forEach((cmd) => {
      expect(validCommands).toContain(cmd);
    });
  });
});
```

### 4. 响应式测试

#### 4.1 Signal 响应式

```typescript
import { signal } from "@type-dom/signals";
import { TdReactiveSvg } from "../src/lib/common/reactive";

describe("Reactivity", () => {
  it("should react to signal changes", () => {
    const colorSignal = signal("red");
    const svg = new TdReactiveSvg({ fill: colorSignal });

    // Initial state
    expect(svg.props.fill.get()).toBe("red");

    // Change signal
    colorSignal.set("blue");
    expect(svg.props.fill.get()).toBe("blue");
  });

  it("should update DOM on signal change", () => {
    const sizeSignal = signal(24);
    const svg = new TdReactiveSvg({ width: sizeSignal });

    const initialWidth = svg.props.width.get();
    expect(initialWidth).toBe(24);

    sizeSignal.set(48);
    const updatedWidth = svg.props.width.get();
    expect(updatedWidth).toBe(48);
  });
});
```

## 🧪 断言技巧

### 常用断言模式

```typescript
import { expect, describe, it } from "vitest";

describe("Assertion Patterns", () => {
  // 1. 相等断言
  it("uses toBe for primitives", () => {
    expect(1 + 1).toBe(2);
    expect("hello").toBe("hello");
  });

  // 2. 对象断言
  it("uses toEqual for objects", () => {
    expect({ a: 1 }).toEqual({ a: 1 });
  });

  // 3. 布尔断言
  it("uses toBeTruthy/toBeFalsy", () => {
    expect(true).toBeTruthy();
    expect(false).toBeFalsy();
  });

  // 4. 数组断言
  it("uses toContain for arrays", () => {
    expect([1, 2, 3]).toContain(2);
    expect([]).toHaveLength(0);
  });

  // 5. 异常断言
  it("uses toThrow for errors", () => {
    expect(() => {
      throw new Error("test");
    }).toThrow();
  });

  // 6. 字符串匹配
  it("uses toMatch for strings", () => {
    expect("hello world").toMatch(/world/);
    expect("viewBox").toMatch(/^\d+ \d+ \d+ \d+$/);
  });

  // 7. 定义检查
  it("uses toBeDefined/toBeUndefined", () => {
    expect({}.nonexistent).toBeUndefined();
    expect({}.hasOwnProperty).toBeDefined();
  });
});
```

### SVG 特定断言

```typescript
describe("SVG Specific Assertions", () => {
  it("validates viewBox format", () => {
    const svg = new TdIconSvg();
    const viewBox = svg.getAttribute("viewBox");
    expect(viewBox).toMatch(/^(\d+\s+){3}\d+$/);
  });

  it("validates path data structure", () => {
    const svg = new TdArrowSvg();
    const path = svg.childNodes[0] as SvgPath;
    const d = path.getData();

    // Path should start with M (Move to)
    expect(d).toMatch(/^M/);

    // Path should contain valid commands
    expect(d).toMatch(/[MLHVCSQTAZ]/);
  });

  it("validates color format", () => {
    const svg = new TdIconSvg({ fill: "#ff0000" });
    expect(svg.props.fill).toMatch(/^#?[0-9a-fA-F]{3,8}$/);
  });
});
```

## 🐛 常见错误和解决方案

### 错误 1: 测试耦合

```typescript
// ❌ WRONG: Tests share state
let sharedSvg: TdIconSvg;

beforeEach(() => {
  sharedSvg = new TdIconSvg();
});

it("test 1 modifies svg", () => {
  sharedSvg.props.width = 100;
});

it("test 2 assumes original state", () => {
  expect(sharedSvg.props.width).toBe(24); // FAILS!
});

// ✅ CORRECT: Independent instances
it("test 1", () => {
  const svg = new TdIconSvg();
  svg.props.width = 100;
});

it("test 2", () => {
  const svg = new TdIconSvg();
  expect(svg.props.width).toBe(24);
});
```

### 错误 2: 断言不足

```typescript
// ❌ WRONG: Insufficient assertions
it("creates svg", () => {
  const svg = new TdIconSvg();
  // No assertions!
});

// ✅ CORRECT: Multiple assertions
it("creates svg with correct properties", () => {
  const svg = new TdIconSvg();
  expect(svg.className).toBe("TdIconSvg");
  expect(svg.childNodes.length).toBeGreaterThan(0);
  expect(svg.getAttribute("viewBox")).toBeDefined();
});
```

### 错误 3: 测试过慢

```typescript
// ❌ WRONG: Slow test
it("loads many icons", () => {
  for (let i = 0; i < 1000; i++) {
    new TdIconSvg();
  }
}); // Takes 2 seconds

// ✅ CORRECT: Representative sample
it("loads icon efficiently", () => {
  const start = performance.now();
  for (let i = 0; i < 10; i++) {
    new TdIconSvg();
  }
  const duration = performance.now() - start;
  expect(duration).toBeLessThan(100); // < 100ms
});
```
