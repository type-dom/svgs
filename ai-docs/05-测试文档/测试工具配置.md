```typescript
      "tests/**/*.spec.ts", // 测试文件匹配模式

  pack: tsdownConfig,
```

```typescript
/// <reference types="vitest" />

import type { Assertion, AsymmetricMatchersContaining } from "vitest";

interface CustomMatchers<R = unknown> {
  toBeSvg(): R;
  toBeValidPath(): R;
  toHaveViewBox(): R;
}

declare module "vitest" {
  interface Assertion<T = any> extends CustomMatchers<T> {}
  interface AsymmetricMatchersContaining extends CustomMatchers {}
}

declare global {
  namespace Vi {
    interface CustomMatchers<T> {
      toBeSvg(): T;
      toBeValidPath(data: string): T;
      toHaveViewBox(expected: string): T;
    }
  }
}

export {};
```

### 自定义 Matcher 实现

```typescript
// tests/setup/custom-matchers.ts
import { expect } from "vitest";
import { TypeSvgSvg, SvgPath } from "@type-dom/framework";

expect.extend({
  toBeSvg(received: any) {
    const pass = received instanceof TypeSvgSvg;
    return {
      pass,
      message: () =>
        `expected ${received?.constructor?.name || received} ${pass ? "not " : ""}to be an SVG instance`,
    };
  },

  toBeValidPath(received: string) {
    if (typeof received !== "string") {
      return {
        pass: false,
        message: () => `expected path data to be a string`,
      };
    }

    // Basic SVG path validation
    const validPathRegex = /^M\s*\d+/;
    const pass = validPathRegex.test(received);

    return {
      pass,
      message: () => `expected path "${received}" ${pass ? "not " : ""}to be a valid SVG path`,
    };
  },

  toHaveViewBox(received: any, expected: string) {
    const viewBox = received.getAttribute?.("viewBox");
    const pass = viewBox === expected;

    return {
      pass,
      message: () => `expected viewBox to be ${expected}, got ${viewBox}`,
    };
  },
});
```

```typescript
// tests/helpers/test-helpers.ts
import { vi } from "vitest";
import { TypeSvgSvg, SvgPath } from "@type-dom/framework";

export class TestHelpers {
  /**
   * 创建 Mock SVG 实例
   */
  static createMockSvg(overrides = {}) {
    return {
      className: "MockSvg",
      childNodes: [],
      props: {
        width: 24,
        height: 24,
        fill: "currentColor",
      },
      addChild: vi.fn(),
      getAttribute: vi.fn().mockReturnValue(""),
      setAttribute: vi.fn(),
      ...overrides,
    };
  }

  /**
   * 创建 Mock Path 实例
   */
  static createMockPath(data = "M0 0 L10 10") {
    return {
      setData: vi.fn(),
      getData: vi.fn().mockReturnValue(data),
      attrObj: {
        fill: "currentColor",
        stroke: "none",
      },
    };
  }

  /**
   * 验证 SVG 基础结构
   */
  static validateSvgStructure(svg: any, expectedClassName: string) {
    expect(svg.className).toBe(expectedClassName);
    expect(Array.isArray(svg.childNodes)).toBe(true);
    expect(svg.getAttribute("viewBox")).toBeDefined();
  }

  /**
   * 验证 Path 数据
   */
  static validatePathData(path: any) {
    const data = path.getData();
    expect(data).toBeDefined();
    expect(typeof data).toBe("string");
    expect(data.length).toBeGreaterThan(0);
    expect(data).toMatch(/^M/);
  }

  /**
   * 性能测试辅助
   */
  static measureExecutionTime(fn: () => void): number {
    const start = performance.now();
    fn();
    const end = performance.now();
    return end - start;
  }
}
```

### 工厂函数

```typescript
// tests/factories/svg-factory.ts
export class SvgFactory {
  /**
   * 创建标准 Props
   */
  static createProps(overrides = {}) {
    return {
      width: 24,
      height: 24,
      fill: "currentColor",
      opacity: 1,
      transform: "",
      title: "",
      ...overrides,
    };
  }

  /**
   * 创建常见颜色值
   */
  static createColor(type: "hex" | "rgb" | "rgba" | "named" = "hex") {
    const colors = {
      hex: ["#ff0000", "#00ff00", "#0000ff"],
      rgb: ["rgb(255,0,0)", "rgb(0,255,0)", "rgb(0,0,255)"],
      rgba: ["rgba(255,0,0,0.5)", "rgba(0,255,0,0.5)", "rgba(0,0,255,0.5)"],
      named: ["red", "green", "blue"],
    };

    const list = colors[type];
    return list[Math.floor(Math.random() * list.length)];
  }

  /**
   * 创建随机尺寸
   */
  static createSize(range: "small" | "medium" | "large" = "medium") {
    const ranges = {
      small: { min: 12, max: 24 },
      medium: { min: 24, max: 48 },
      large: { min: 48, max: 96 },
    };

    const { min, max } = ranges[range];
    const size = Math.floor(Math.random() * (max - min + 1)) + min;

    return { width: size, height: size };
  }
}
```

### 覆盖率脚本

```typescript
// scripts/check-coverage.ts
import { readFileSync } from "fs";
import { join } from "path";

interface CoverageSummary {
  total: {
    lines: { pct: number };
    functions: { pct: number };
    branches: { pct: number };
    statements: { pct: number };
  };
}

const THRESHOLDS = {
  lines: 90,
  functions: 95,
  branches: 85,
  statements: 90,
};

function checkCoverage() {
  try {
    const summaryPath = join(process.cwd(), "coverage", "coverage-summary.json");
    const summary: CoverageSummary = JSON.parse(readFileSync(summaryPath, "utf-8"));

    const checks = [
      { name: "Lines", actual: summary.total.lines.pct, threshold: THRESHOLDS.lines },
      { name: "Functions", actual: summary.total.functions.pct, threshold: THRESHOLDS.functions },
      { name: "Branches", actual: summary.total.branches.pct, threshold: THRESHOLDS.branches },
      {
        name: "Statements",
        actual: summary.total.statements.pct,
        threshold: THRESHOLDS.statements,
      },
    ];

    let allPassed = true;

    console.log("\n📊 Coverage Report\n");
    console.log("=".repeat(60));

    checks.forEach((check) => {
      const passed = check.actual >= check.threshold;
      const icon = passed ? "✅" : "❌";
      const status = passed ? "PASS" : "FAIL";

      console.log(
        `${icon} ${check.name}: ${check.actual.toFixed(2)}% (threshold: ${check.threshold}%) - ${status}`,
      );

      if (!passed) {
        allPassed = false;
      }
    });

    console.log("=".repeat(60));

    if (!allPassed) {
      console.error("\n❌ Coverage thresholds not met!\n");
      process.exit(1);
    } else {
      console.log("\n✅ All coverage thresholds met!\n");
      process.exit(0);
    }
  } catch (error) {
    console.error("Failed to check coverage:", error);
    process.exit(1);
  }
}

checkCoverage();
```
