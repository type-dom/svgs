## 🎯 集成测试定义

### 与单元测试的区别

| 维度          | 单元测试     | 集成测试             |
| ------------- | ------------ | -------------------- |
| **测试对象**  | 单个组件     | 多个组件组合         |
| **测试范围**  | 独立功能     | 交互和接口           |
| **Mock 使用** | 大量 Mock    | 少量 Mock 或无 Mock  |
| **执行速度**  | 快 (< 100ms) | 中等 (100ms - 1s)    |
| **发现问题**  | 逻辑错误     | 接口不匹配、集成问题 |

### TypeDOM SVG 集成测试场景

```typescript
// 单元测试：测试单个 SVG
import { TdUserSvg } from "../src/lib/common/user";

describe("TdUserSvg Unit", () => {
  it("should create instance", () => {
    const svg = new TdUserSvg();
    expect(svg.className).toBe("TdUserSvg");
  });
});

// 集成测试：测试多个 SVG 组合
import { TypeDiv } from "@type-dom/framework";
import { TdUserSvg, TdAddSvg } from "../src/lib/common-index";

describe("SVG Integration", () => {
  it("should render multiple SVGs together", () => {
    const container = new TypeDiv();
    const userIcon = new TdUserSvg();
    const addIcon = new TdAddSvg();

    container.addChild(userIcon, addIcon);

    expect(container.childNodes.length).toBe(2);
    expect(container.childNodes[0]).toBeInstanceOf(TdUserSvg);
    expect(container.childNodes[1]).toBeInstanceOf(TdAddSvg);
  });
});
```

## 🔧 组件集成测试

### 1. 父子组件集成

```typescript
import { describe, it, expect, beforeEach } from "vitest";
import { TypeDiv, TypeSvgSvg } from "@type-dom/framework";
import { TdUserSvg, TdAddSvg, TdCloseSvg } from "../../src/lib/common-index";

describe("Parent-Child Integration", () => {
  let container: TypeDiv;
  let icons: TypeSvgSvg[];

  beforeEach(() => {
    container = new TypeDiv();
    icons = [new TdUserSvg(), new TdAddSvg(), new TdCloseSvg()];
  });

  describe("组件添加", () => {
    it("✅ 应该能添加多个子组件", () => {
      icons.forEach((icon) => container.addChild(icon));

      expect(container.childNodes.length).toBe(3);
    });

    it("✅ 应该保持添加顺序", () => {
      container.addChild(icons[0], icons[1], icons[2]);

      expect(container.childNodes[0]).toBe(icons[0]);
      expect(container.childNodes[1]).toBe(icons[1]);
      expect(container.childNodes[2]).toBe(icons[2]);
    });

    it("✅ 应该支持动态添加", () => {
      container.addChild(icons[0]);
      expect(container.childNodes.length).toBe(1);

      container.addChild(icons[1]);
      expect(container.childNodes.length).toBe(2);
    });
  });

  describe("样式继承", () => {
    it("✅ 子组件应该继承父组件的颜色", () => {
      container.props.style = { color: "blue" };
      const icon = new TdUserSvg();
      container.addChild(icon);

      // Icon uses currentColor
      expect(icon.props.fill).toBe("currentColor");
    });

    it("✅ 子组件可以覆盖继承样式", () => {
      container.props.style = { color: "blue" };
      const icon = new TdUserSvg({ fill: "red" });
      container.addChild(icon);

      expect(icon.props.fill).toBe("red");
    });
  });

  describe("事件传播", () => {
    it("✅ 应该支持事件冒泡", () => {
      const clickHandler = vi.fn();
      container.props.onClick = clickHandler;

      const icon = new TdUserSvg({
        onClick: () => {},
      });
      container.addChild(icon);

      // Simulate click on icon
      icon.props.onClick?.();

      // Event should bubble to parent
      expect(clickHandler).toHaveBeenCalled();
    });
  });
});
```

### 2. 兄弟组件交互

```typescript
describe("Sibling Interaction", () => {
  it("✅ 应该支持兄弟组件同步状态", () => {
    const container = new TypeDiv();

    const icon1 = new TdUserSvg({ opacity: 1 });
    const icon2 = new TdAddSvg({ opacity: 1 });

    // Setup interaction
    icon1.props.onClick = () => {
      icon2.props.opacity = 0.5;
    };

    container.addChild(icon1, icon2);

    // Trigger interaction
    icon1.props.onClick?.();

    expect(icon2.props.opacity).toBe(0.5);
  });

  it("✅ 应该支持条件渲染", () => {
    const container = new TypeDiv();
    const showIcon = true;

    if (showIcon) {
      const icon = new TdUserSvg();
      container.addChild(icon);
    }

    expect(container.childNodes.length).toBe(1);
  });
});
```

## 🌐 服务集成测试

### 1. 与外部库集成

```typescript
import { describe, it, expect } from "vitest";
import { signal } from "@type-dom/signals";
import { TdReactiveSvg } from "../../src/lib/common/reactive";

describe("External Library Integration", () => {
  describe("Signals 集成", () => {
    it("✅ 应该响应 Signal 变化", () => {
      const colorSignal = signal("red");
      const svg = new TdReactiveSvg({ fill: colorSignal });

      // Initial state
      expect(svg.props.fill.get()).toBe("red");

      // Change signal
      colorSignal.set("blue");
      expect(svg.props.fill.get()).toBe("blue");
    });

    it("✅ 应该支持 Computed Signal", () => {
      const sizeSignal = signal(24);
      const doubleSize = signal(sizeSignal.get() * 2);

      const svg = new TdReactiveSvg({ width: sizeSignal });

      expect(svg.props.width.get()).toBe(24);

      sizeSignal.set(48);
      // Note: In real implementation, computed would update automatically
      expect(doubleSize.get()).toBe(96);
    });
  });

  describe("Effect 集成", () => {
    it("✅ 应该触发 Effect", () => {
      const countSignal = signal(0);
      const effectFn = vi.fn();

      // Setup effect
      const stop = effect(() => {
        effectFn(countSignal.get());
      });

      expect(effectFn).toHaveBeenCalledWith(0);

      countSignal.set(1);
      expect(effectFn).toHaveBeenCalledWith(1);

      stop();
    });
  });
});
```

### E2E 测试示例

```typescript
// tests/e2e/svg-rendering.spec.ts
import { test, expect } from "@playwright/test";

test.describe("SVG Rendering E2E", () => {
  test("✅ 应该正确渲染 User 图标", async ({ page }) => {
    await page.goto("/");

    // Find SVG element
    const svgElement = page.locator('[data-testid="user-icon"]');
    await expect(svgElement).toBeVisible();

    // Check viewBox
    const viewBox = await svgElement.getAttribute("viewBox");
    expect(viewBox).toBe("0 0 128 128");

    // Check rendering
    const svgBox = await svgElement.boundingBox();
    expect(svgBox?.width).toBe(24);
    expect(svgBox?.height).toBe(24);
  });

  test("✅ 应该支持点击交互", async ({ page }) => {
    await page.goto("/");

    const icon = page.locator('[data-testid="add-icon"]');
    await icon.click();

    // Verify click handler was called
    await expect(page.locator("#click-count")).toHaveText("1");
  });

  test("✅ 应该支持主题切换", async ({ page }) => {
    await page.goto("/");

    // Switch to dark mode
    await page.click('[data-testid="dark-mode-toggle"]');

    // Check SVG color changed
    const fillColor = await page.evaluate(() => {
      const svg = document.querySelector('[data-testid="user-icon"] path');
      return window.getComputedStyle(svg!).fill;
    });

    expect(fillColor).toBe("rgb(255, 255, 255)"); // White in dark mode
  });
});
```

```typescript
import { signal } from "@type-dom/signals";

describe("Real-world Scenario: Dynamic Theme", () => {
  it("✅ 应该响应主题变化", () => {
    const themeSignal = signal<"light" | "dark">("light");

    const getThemeColor = () => {
      return themeSignal.get() === "light" ? "#000000" : "#ffffff";
```

## ⚠️ 常见集成问题和解决方案

### 问题 1: Props 传递不正确

```typescript
// ❌ WRONG: Props not passed correctly
const parent = new TypeDiv();
const child = new TdUserSvg();
parent.addChild(child);
// Child doesn't inherit parent's style

// ✅ CORRECT: Explicit prop passing
const parent = new TypeDiv({
  style: { color: "blue" },
});
const child = new TdUserSvg({
  fill: "currentColor", // Use currentColor to inherit
});
parent.addChild(child);
```

### 问题 2: 事件处理冲突

```typescript
// ❌ WRONG: Event handler conflict
const parent = new TypeDiv({
  onClick: () => console.log("parent"),
});
const child = new TdUserSvg({
  onClick: () => console.log("child"), // Prevents parent handler
});

// ✅ CORRECT: Event bubbling
const parent = new TypeDiv({
  onClick: () => console.log("parent"),
});
const child = new TdUserSvg({
  onClick: (e) => {
    console.log("child");
    e?.bubble(); // Allow event to bubble
  },
});
```

### 问题 3: 样式覆盖问题

```typescript
// ❌ WRONG: Style override issue
const parent = new TypeDiv({
  style: { color: "blue" },
});
const child = new TdUserSvg({
  fill: "red", // Hard override, no inheritance
});

// ✅ CORRECT: Conditional styling
const parent = new TypeDiv({
  style: { color: "blue" },
});
const child = new TdUserSvg({
  fill: "currentColor", // Inherits, can be overridden if needed
});
```
