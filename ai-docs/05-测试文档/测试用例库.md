## 🔢 Common 分类测试用例

### 基础图标测试

#### 1. Td404Svg 测试

```typescript
/**
 * @file Td404Svg 组件完整测试
 */
import { describe, it, expect, beforeEach } from 'vitest';
import { Td404Svg } from '../../src/lib/common/404';

describe('Td404Svg', () => {
  let svg: Td404Svg;

  beforeEach(() => {
    svg = new Td404Svg();
  });

  describe('基础属性', () => {
    it('✅ 应该有正确的 className', () => {
      expect(svg.className).toBe('Td404Svg');
    });

    it('✅ 应该有 childNodes 数组', () => {
      expect(Array.isArray(svg.childNodes)).toBe(true);
      expect(svg.childNodes.length).toBeGreaterThan(0);
    });

    it('✅ 应该有 viewBox 属性', () => {
      const viewBox = svg.getAttribute('viewBox');
      expect(viewBox).toBe('0 0 128 128');
    });

    it('✅ 应该有默认尺寸 24x24', () => {
      expect(svg.props.width).toBe(24);
      expect(svg.props.height).toBe(24);
    });
  });

  describe('Path 验证', () => {
    it('✅ 应该包含至少一个 Path', () => {
      expect(svg.childNodes.length).toBeGreaterThanOrEqual(1);
    });

    it('✅ Path 应该使用 currentColor 填充', () => {
      const path = svg.childNodes[0];
      expect(path.attrObj.fill).toBe('currentColor');
    });

    it('✅ Path 数据应该有效', () => {
      const path = svg.childNodes[0];
      const d = path.getData();
      
      expect(d).toBeDefined();
      expect(typeof d).toBe('string');
      expect(d.length).toBeGreaterThan(0);
      expect(d).toMatch(/^M/); // Should start with Move command
    });
  });

  describe('自定义 Props', () => {
    it('✅ 应该接受自定义尺寸', () => {
      const customSvg = new Td404Svg({ width: 32, height: 32 });
      expect(customSvg.props.width).toBe(32);
      expect(customSvg.props.height).toBe(32);
    });

    it('✅ 应该接受自定义颜色', () => {
      const customSvg = new Td404Svg({ fill: 'red' });
      expect(customSvg.props.fill).toBe('red');
    });

    it('✅ 应该接受多个 Props', () => {
      const customSvg = new Td404Svg({
        width: 48,
        height: 48,
        fill: 'blue',
        opacity: 0.8
      });
      
      expect(customSvg.props.width).toBe(48);
      expect(customSvg.props.fill).toBe('blue');
      expect(customSvg.props.opacity).toBe(0.8);
    });
  });

  describe('继承关系', () => {
    it('✅ 应该继承自 TypeSvgSvg', () => {
      expect(svg instanceof (globalThis as any).TypeSvgSvg).toBe(true);
    });

    it('✅ 应该继承 TypeSvgSvg 的方法', () => {
      expect(typeof svg.addChild).toBe('function');
      expect(typeof svg.getAttribute).toBe('function');
    });
  });
});
```

```typescript
      expect(svg.className).toBe('FluentArrowSvg');
    });

    it('✅ FluentUI Check 图标应该存在', () => {
      expect(FluentUIIcons.FluentCheckSvg).toBeDefined();
      const svg = new FluentUIIcons.FluentCheckSvg();
      expect(svg.className).toBe('FluentCheckSvg');
    });
  });

  describe('批量验证', () => {
    it('✅ 所有导出组件应该有 className', () => {
      const components = Object.values(FluentUIIcons);
      components.forEach(comp => {
        if (typeof comp === 'function') {
          const instance = new (comp as any)();
          expect(instance.className).toBeDefined();
        }
      });
    });

    it('✅ 随机抽样 10 个图标验证', () => {
      const componentNames = Object.keys(FluentUIIcons);
      const randomSamples = componentNames
        .sort(() => Math.random() - 0.5)
        .slice(0, 10);

      randomSamples.forEach(name => {
        const Component = (FluentUIIcons as any)[name];
        if (typeof Component === 'function') {
          const instance = new Component();
          expect(instance.className).toBeDefined();
          expect(instance.getAttribute('viewBox')).toBeDefined();
        }
      });
    });
  });
});
```

## 🚀 性能测试用例

### 实例化性能

```typescript
import { describe, it, expect, bench } from 'vitest';
import { TdIconSvg } from '../../src/lib/common/icon';

describe('Performance Tests', () => {
  
  describe('实例化性能', () => {
    bench('创建单个图标', () => {
      new TdIconSvg();
    }, {
      iterations: 1000,
      warmupIterations: 100
    });

    bench('批量创建 100 个图标', () => {
      for (let i = 0; i < 100; i++) {
        new TdIconSvg();
      }
    }, {
      iterations: 10,
      warmupIterations: 2
    });
  });

  describe('内存使用', () => {
    it('✅ 单个图标内存占用应合理', () => {
      const beforeMemory = process.memoryUsage();
      const svg = new TdIconSvg();
      const afterMemory = process.memoryUsage();
      
      const memoryDiff = afterMemory.heapUsed - beforeMemory.heapUsed;
      expect(memoryDiff).toBeLessThan(1024 * 1024); // < 1MB
    });

    it('✅ 销毁后应释放内存', () => {
      const svgs = [];
      for (let i = 0; i < 100; i++) {
        svgs.push(new TdIconSvg());
      }
      
      // Clear references
      svgs.length = 0;
      
      // Memory should be reclaimable
      global.gc?.();
    });
  });
});
```
