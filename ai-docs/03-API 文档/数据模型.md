## 🔧 核心类型定义

### 1. SvgProps 类型

SVG 组件属性的完整类型定义:

```typescript
/**
 * SVG 组件属性接口
 * 
 * @example
 * ```typescript
 * const props: SvgProps = {
 *   width: 24,
 *   height: 24,
 *   viewBox: '0 0 1024 1024',
 *   fill: 'currentColor',
 *   onClick: handleClick
 * };
 * ```
 */
interface SvgProps {
  // ========== 尺寸属性 ==========
  /**
   * SVG 宽度
   * - 数字：表示像素
   * - 字符串：支持 CSS 单位（%, em, rem, px 等）
   * 
   * @example
   * width: 24           // 24 像素
   * width: "100%"       // 父容器宽度的 100%
   * width: "2em"        // 2 倍字体大小
   * width: "1.5rem"     // 1.5 倍根元素字体大小
   */
  width?: number | string;
  
  /**
   * SVG 高度
   * - 数字：表示像素
   * - 字符串：支持 CSS 单位
   */
  height?: number | string;
  
  /**
   * viewBox 属性
   * 格式："min-x min-y width height"
   * 
   * @default "0 0 1024 1024"
   * 
   * @example
   * viewBox: "0 0 1024 1024"  // 标准坐标系
   * viewBox: "0 0 24 24"      // 小图标坐标系
   * viewBox: "0 0 512 512"    // 中等坐标系
   */
  viewBox?: string;
  
  // ========== 填充和描边 ==========
  /**
   * 填充颜色
   * 
   * 支持的颜色格式:
   * - 十六进制：#RGB, #RRGGBB, #RGBA, #RRGGBBAA
   * - RGB/RGBA: rgb(255, 0, 0), rgba(255, 0, 0, 0.5)
   * - HSL/HSLA: hsl(0, 100%, 50%), hsla(0, 100%, 50%, 0.5)
   * - 颜色名称：red, blue, green 等
   * - currentColor: 继承父元素的 color 属性
   * - transparent: 透明
   * - none: 无填充
   * 
   * @default "currentColor"
   * 
   * @example
   * fill: "#ff0000"         // 红色
   * fill: "rgb(0, 255, 0)"  // 绿色
   * fill: "currentColor"    // 继承颜色
   * fill: "transparent"     // 透明填充
   * fill: "none"            // 无填充
   */
  fill?: string;
  
  /**
   * 描边颜色
   * 
   * 支持格式同 fill
   * 
   * @default "none"
   * 
   * @example
   * stroke: "#000000"       // 黑色描边
   * stroke: "currentColor"  // 继承颜色
   * stroke: "none"          // 无描边（默认）
   */
  stroke?: string;
  
  /**
   * 描边宽度
   * - 数字：表示像素
   * - 字符串：支持 CSS 单位
   * 
   * @default 0
   * 
   * @example
   * strokeWidth: 1          // 1 像素
   * strokeWidth: "2px"      // 2 像素
   * strokeWidth: "0.5pt"    // 0.5 点
   */
  strokeWidth?: number | string;
  
  /**
   * 透明度
   * 范围：0.0 (完全透明) - 1.0 (完全不透明)
   * 
   * @default 1.0
   * 
   * @example
   * opacity: 1.0    // 完全不透明
   * opacity: 0.5    // 半透明
   * opacity: 0.0    // 完全透明
   */
  opacity?: number;
  
  // ========== 事件处理器 ==========
  /**
   * 点击事件处理器
   * 
   * @param event - MouseEvent 对象
   * 
   * @example
   * onClick: (event: MouseEvent) => {
   *   console.log('Clicked!', event.target);
   * }
   */
  onClick?: (event: MouseEvent) => void;
  
  /**
   * 鼠标移入事件处理器
   * 
   * @example
   * onMouseEnter: () => setHovered(true)
   */
  onMouseEnter?: (event: MouseEvent) => void;
  
  /**
   * 鼠标移出事件处理器
   * 
   * @example
   * onMouseLeave: () => setHovered(false)
   */
  onMouseLeave?: (event: MouseEvent) => void;
  
  /**
   * 鼠标按下事件处理器
   */
  onMouseDown?: (event: MouseEvent) => void;
  
  /**
   * 鼠标松开事件处理器
   */
  onMouseUp?: (event: MouseEvent) => void;
  
  /**
   * 鼠标移动事件处理器
   */
  onMouseMove?: (event: MouseEvent) => void;
  
  /**
   * 双击事件处理器
   */
  onDblClick?: (event: MouseEvent) => void;
  
  /**
   * 键盘按下事件处理器
   */
  onKeyDown?: (event: KeyboardEvent) => void;
  
  /**
   * 键盘松开事件处理器
   */
  onKeyUp?: (event: KeyboardEvent) => void;
  
  /**
   * 焦点获得事件处理器
   */
  onFocus?: (event: FocusEvent) => void;
  
  /**
   * 焦点失去事件处理器
   */
  onBlur?: (event: FocusEvent) => void;
  
  // ========== 其他 SVG 属性 ==========
  /**
   * 保持宽高比
   * 
   * 常用值:
   * - "xMidYMid meet": 居中缩放以保持适配（默认）
   * - "xMidYMid slice": 居中裁剪以填满视图
   * - "none": 不保持宽高比
   * 
   * @default "xMidYMid meet"
   * 
   * @example
   * preserveAspectRatio: "xMidYMid meet"
   * preserveAspectRatio: "xMinYMin slice"
   * preserveAspectRatio: "none"
   */
  preserveAspectRatio?: string;
  
  /**
   * XML 命名空间
   * 
   * @default "http://www.w3.org/2000/svg"
   */
  xmlns?: string;
  
  /**
   * 唯一标识符
   * 
   * @example
   * id: "add-icon-123"
   */
  id?: string;
  
  /**
   * CSS 类名
   * 
   * @example
   * class: "icon icon-add active"
   */
  class?: string;
  
  /**
   * 内联样式（字符串格式）
   * 
   * @example
   * style: "display: block; margin: 10px;"
   */
  style?: string;
  
  /**
   * 标题文本（用于无障碍访问）
   * 
   * @example
   * title: "Add Icon"
   */
  title?: string;
  
  /**
   * 描述文本（用于无障碍访问）
   * 
   * @example
   * desc: "A plus sign icon used for adding items"
   */
  desc?: string;
}
```

### 2. TypeProps 类型

TypeDom 通用组件属性:

```typescript
/**
 * TypeDom 通用组件属性接口
 * 
 * 所有 TypeDom 组件都支持这些属性
 */
interface TypeProps {
  /**
   * 组件类名（用于 TypeScript 类型识别）
   * 
   * @example
   * className: 'TdAddSvg'
   */
  className?: string;
  
  /**
   * DOM 元素 ID
   */
  id?: string;
  
  /**
   * HTML 属性对象
   * 
   * @example
   * attrObj: {
   *   role: "img",
   *   "aria-label": "Add icon",
   *   tabindex: "0"
   * }
   */
  attrObj?: Record<string, any>;
  
  /**
   * CSS 样式对象
   * 
   * @example
   * styleObj: {
   *   display: "block",
   *   margin: "10px auto",
   *   cursor: "pointer"
   * }
   */
  styleObj?: Record<string, any>;
  
  /**
   * 子组件数组
   * 
   * @example
   * children: [
   *   new SvgPath({ attrObj: { fill: 'currentColor' }}),
   *   new SvgTitle({ text: 'Icon Title' })
   * ]
   */
  children?: TypeElement[];
  
  /**
   * 事件处理器对象
   * 
   * @example
   * events: {
   *   onClick: handleClick,
   *   onMouseEnter: handleMouseEnter
   * }
   */
  events?: Record<string, Function>;
  
  /**
   * 引用（用于访问组件实例）
   * 
   * @example
   * ref: (component) => { this.iconRef = component; }
   */
  ref?: any;
}
```

### 3. PathData 类型

SVG 路径数据类型:

```typescript
/**
 * SVG 路径数据
 * 
 * 路径命令格式:
 * - M/m: MoveTo (移动到)
 * - L/l: LineTo (画线到)
 * - H/h: Horizontal LineTo (水平画线)
 * - V/v: Vertical LineTo (垂直画线)
 * - C/c: Cubic Bezier Curve (三次贝塞尔曲线)
 * - S/s: Smooth Cubic Bezier (平滑三次贝塞尔)
 * - Q/q: Quadratic Bezier Curve (二次贝塞尔曲线)
 * - T/t: Smooth Quadratic Bezier (平滑二次贝塞尔)
 * - A/a: Arc (椭圆弧)
 * - Z/z: ClosePath (闭合路径)
 * 
 * 大写字母表示绝对坐标，小写字母表示相对坐标
 */
type PathData = string;

/**
 * 路径数据示例
 * 
 * @example
 * ```typescript
 * // 一个简单的正方形路径
 * const squarePath: PathData = `
 *   M 10 10
 *   L 90 10
 *   L 90 90
 *   L 10 90
 *   Z
 * `;
 * 
 * // 圆形路径（使用贝塞尔曲线近似）
 * const circlePath: PathData = `
 *   M 50 0
 *   C 77.6 0 100 22.4 100 50
 *   C 100 77.6 77.6 100 50 100
 *   C 22.4 100 0 77.6 0 50
 *   C 0 22.4 22.4 0 50 0
 *   Z
 * `;
 * 
 * // 加号图标路径
 * const addIconPath: PathData = `
 *   M512 42.666667c259.2 0 469.333333 210.133333 469.333333 469.333333
 *   S771.2 981.333333 512 981.333333 42.666667 771.2 42.666667 512 210.133333 42.666667 512 42.666667z
 *   m234.666667 492.8H535.466667v211.2c0 12.8-10.666667 23.466667-23.466667 23.466667
 *   s-23.466667-10.666667-23.466667-23.466667v-211.2H277.333333c-12.8 0-23.466667-10.666667
 *   -23.466667-23.466667s10.666667-23.466667 23.466667-23.466667h211.2V277.333333
 *   c0-12.8 10.666667-23.466667 23.466667-23.466667s23.466667 10.666667 23.466667 23.466667
 *   v211.2h211.2c12.8 0 23.466667 10.666667 23.466667 23.466667s-10.666667 23.466667
 *   -23.466667 23.466667z
 * `;
 * ```
 */
```

## 🏗️ 类定义

### 1. TypeSvgSvg 类

SVG 组件的基类:

```typescript
/**
 * TypeSvgSvg - SVG 根元素基类
 * 
 * 所有 SVG 图标组件都继承自此类
 * 
 * @extends TypeSvg
 * 
 * @example
 * ```typescript
 * export class TdAddSvg extends TypeSvgSvg {
 *   className: 'TdAddSvg';
 *   override childNodes: SvgPath[];
 *   
 *   constructor(params: SvgProps = {}) {
 *     super(params);
 *     // ... 实现
 *   }
 * }
 * ```
 */
declare class TypeSvgSvg extends TypeSvg {
  /**
   * 组件类名（必须实现）
   * 
   * 用于 TypeScript 类型识别和运行时类型检查
   */
  className: string;
  
  /**
   * 子节点数组（重写为 SvgPath 类型）
   * 
   * SVG 组件的子节点通常是 SvgPath 实例
   */
  override childNodes: SvgPath[];
  
  /**
   * 构造函数
   * 
   * @param params - SVG 属性参数
   */
  constructor(params?: SvgProps);
  
  /**
   * 重置 SVG 尺寸
   * 
   * @param width - 宽度
   * @param height - 高度
   * 
   * @example
   * ```typescript
   * const icon = new TdAddSvg();
   * icon.resetSize(32, 32);  // 设置为 32x32 像素
   * icon.resetSize("2em", "2em");  // 设置为 2em x 2em
   * ```
   */
  resetSize(width: number | string, height: number | string): void;
  
  /**
   * 添加子组件
   * 
   * @param children - 子组件数组
   * 
   * @example
   * ```typescript
   * const icon = new TdAddSvg();
   * const path = new SvgPath({ attrObj: { fill: 'currentColor' }});
   * path.setData('M512 42.666667...');
   * icon.addChild(path);
   * ```
   */
  addChild(...children: TypeElement[]): void;
  
  /**
   * 移除子组件
   * 
   * @param child - 要移除的子组件
   */
  removeChild(child: TypeElement): void;
  
  /**
   * 设置属性
   * 
   * @param key - 属性名
   * @param value - 属性值
   * 
   * @example
   * ```typescript
   * icon.setAttr('fill', '#ff0000');
   * icon.setAttr('viewBox', '0 0 512 512');
   * ```
   */
  setAttr(key: string, value: any): void;
  
  /**
   * 批量设置属性
   * 
   * @param attrs - 属性对象
   * 
   * @example
   * ```typescript
   * icon.setAttrs({
   *   width: 32,
   *   height: 32,
   *   fill: 'blue'
   * });
   * ```
   */
  setAttrs(attrs: Record<string, any>): void;
  
  /**
   * 获取属性值
   * 
   * @param key - 属性名
   * @returns 属性值
   */
  getAttr(key: string): any;
  
  /**
   * 移除属性
   * 
   * @param key - 属性名
   */
  removeAttr(key: string): void;
  
  /**
   * 设置样式
   * 
   * @param key - 样式名
   * @param value - 样式值
   * 
   * @example
   * ```typescript
   * icon.setStyle('opacity', '0.5');
   * icon.setStyle('transform', 'rotate(45deg)');
   * ```
   */
  setStyle(key: string, value: string): void;
  
  /**
   * 批量设置样式
   * 
   * @param styles - 样式对象
   */
  setStyles(styles: Record<string, string>): void;
  
  /**
   * 获取样式值
   * 
   * @param key - 样式名
   * @returns 样式值
   */
  getStyle(key: string): string;
  
  /**
   * 移除样式
   * 
   * @param key - 样式名
   */
  removeStyle(key: string): void;
  
  /**
   * 挂载到 DOM
   * 
   * @param container - 容器元素
   */
  mount(container: HTMLElement): void;
  
  /**
   * 卸载
   */
  unmount(): void;
  
  /**
   * 更新组件
   */
  update(): void;
}
```

### 2. SvgPath 类

SVG 路径组件:

```typescript
/**
 * SvgPath - SVG 路径组件
 * 
 * 用于表示 SVG 的 <path> 元素
 * 
 * @extends TypeElement
 * 
 * @example
 * ```typescript
 * const path = new SvgPath({
 *   attrObj: {
 *     fill: 'currentColor',
 *     stroke: '#000',
 *     strokeWidth: 1
 *   }
 * });
 * path.setData('M512 42.666667...');
 * icon.addChild(path);
 * ```
 */
declare class SvgPath extends TypeElement {
  /**
   * 构造函数
   * 
   * @param params - 路径属性
   */
  constructor(params?: TypeProps);
  
  /**
   * 设置路径数据
   * 
   * @param data - 路径数据字符串
   * 
   * @example
   * ```typescript
   * path.setData('M10 10 L90 90 L90 10 Z');
   * ```
   */
  setData(data: PathData): void;
  
  /**
   * 获取路径数据
   * 
   * @returns 路径数据字符串
   */
  getData(): PathData;
  
  /**
   * 清空路径数据
   */
  clearData(): void;
  
  /**
   * 获取路径长度
   * 
   * @returns 路径长度（像素）
   */
  getTotalLength(): number;
  
  /**
   * 获取路径上的点
   * 
   * @param distance - 距离起点的长度
   * @returns 点的坐标 {x, y}
   */
  getPointAtLength(distance: number): { x: number; y: number };
  
  /**
   * 获取路径在指定位置的切线角度
   * 
   * @param distance - 距离起点的长度
   * @returns 角度（弧度）
   */
  getAngleAtLength(distance: number): number;
}
```

## 🎯 类型使用示例

### 示例 1: 创建自定义 SVG 组件

```typescript
import { SvgProps, SvgPath, TypeSvgSvg, addAttrObj } from '@type-dom/framework';

// ✅ 完整的类型定义
export class TdCustomIconSvg extends TypeSvgSvg {
  // 明确的类型声明
  className: 'TdCustomIconSvg';
  override childNodes: SvgPath[];
  
  // 类型安全的构造函数
  constructor(params: SvgProps = {}) {
    super(params);
    
    this.className = 'TdCustomIconSvg';
    
    // 类型安全的属性添加
    addAttrObj(this, {
      name: 'TdCustomIconSvg',
      viewBox: '0 0 1024 1024'
    });
    
    this.resetSize(24, 24);
    this.childNodes = [];
    
    // 类型安全的路径创建
    const path = new SvgPath({ 
      attrObj: { fill: 'currentColor' }
    });
    
    path.setData('M512...');
    this.addChild(path);
  }
}
```

### 示例 2: 类型安全的组件使用

```typescript
import { TdAddSvg } from '@type-dom/svgs/common/add';

// ✅ 类型推断
const icon = new TdAddSvg();
// icon 的类型：TdAddSvg

// ✅ 类型检查
icon.className;  // 类型：string
icon.childNodes; // 类型：SvgPath[]

// ✅ 类型安全的参数
const iconWithProps = new TdAddSvg({
  width: 32,        // ✅ 正确：number
  height: "2em",    // ✅ 正确：string
  fill: "#ff0000",  // ✅ 正确：string
  opacity: 0.8,     // ✅ 正确：number
  onClick: (e) => { // ✅ 正确：MouseEvent 处理器
    console.log(e.target);
  }
});

// ❌ 类型错误（会被 TypeScript 捕获）
const wrongIcon = new TdAddSvg({
  width: true,      // ❌ 错误：boolean 不是 number | string
  opacity: "high"   // ❌ 错误：string 不是 number
});
```

// ✅ 使用泛型约束确保类型安全
function createIcon<T extends TypeSvgSvg>(
  IconClass: new (params?: SvgProps) => T,
  size: number = 24
): T {
  const icon = new IconClass({
    width: size,
    height: size,
    fill: 'currentColor'
  });
  
  return icon;  // 返回类型：T

## 📊 类型关系图

```
TypeScript 类型系统
│
├─ 基础类型
│  ├─ string
│  ├─ number
│  ├─ boolean
│  └─ any
│
├─ TypeDom 类型
│  ├─ TypeProps
│  │  └─ 包含：className, attrObj, styleObj, children, events
│  │
│  ├─ TypeElement (抽象类)

│  │  └─ 方法：addChild, removeChild, setAttr, setStyle, mount
│  │
│  └─ TypeSvg (抽象类)
│     └─ 扩展：viewBox, preserveAspectRatio, resetSize
│
└─ SVG 专用类型
   ├─ SvgProps (接口)
   │  └─ 扩展 TypeProps，添加 SVG 特定属性
   │
   ├─ TypeSvgSvg (类)
   │  ├─ 继承：TypeSvg
   │  └─ 实现：SvgProps 接口
   │
   ├─ SvgPath (类)
   │  ├─ 继承：TypeElement
   │  └─ 方法：setData, getData, getTotalLength
   │
   └─ PathData (类型别名)
      └─ string (SVG 路径数据)
```
