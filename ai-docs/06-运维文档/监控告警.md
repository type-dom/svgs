| **类型覆盖度**   | >95%   | 90-95% | <90%   | 每周     |

**自动化检查脚本**:

```bash
#!/bin/bash
# scripts/monitor-types.sh

echo "🔍 Running TypeScript type check..."

# 运行类型检查并捕获输出
TYPECHECK_OUTPUT=$(npm run typecheck 2>&1)
EXIT_CODE=$?

if [ $EXIT_CODE -ne 0 ]; then
  ERROR_COUNT=$(echo "$TYPECHECK_OUTPUT" | grep -c "error TS")

  echo "❌ Found $ERROR_COUNT type errors"

  if [ $ERROR_COUNT -gt 5 ]; then
    echo "🚨 CRITICAL: Too many type errors!"
    # 发送告警通知
    send_alert "TypeScript Errors" "$ERROR_COUNT errors found"
    exit 1
  elif [ $ERROR_COUNT -gt 0 ]; then
    echo "⚠️  WARNING: Type errors detected"
    send_warning "TypeScript Warnings" "$ERROR_COUNT errors found"
    exit 1
  fi
else
  echo "✅ No type errors found"
fi

# 检查 any 类型使用
ANY_USAGE=$(echo "$TYPECHECK_OUTPUT" | grep -c "implicitly has an 'any' type" || true)
if [ $ANY_USAGE -gt 0 ]; then
  echo "⚠️  Found $ANY_USAGE implicit any types"
  # 记录但不阻止（可配置为阻止）
fi

exit 0
```

**覆盖率监控脚本**:

```bash
#!/bin/bash
# scripts/monitor-coverage.sh

echo "📊 Analyzing test coverage..."

# 运行测试生成覆盖率报告
npm run test:coverage

# 解析覆盖率 JSON
COVERAGE_FILE="coverage/coverage-final.json"

if [ ! -f "$COVERAGE_FILE" ]; then
  echo "❌ Coverage file not found!"
  exit 1
fi

# 提取各项指标
STATEMENTS=$(cat $COVERAGE_FILE | jq '.total.statements.pct')
BRANCHES=$(cat $COVERAGE_FILE | jq '.total.branches.pct')
FUNCTIONS=$(cat $COVERAGE_FILE | jq '.total.functions.pct')
LINES=$(cat $COVERAGE_FILE | jq '.total.lines.pct')

echo "Coverage Report:"
echo "  Statements: ${STATEMENTS}%"
echo "  Branches:   ${BRANCHES}%"
echo "  Functions:  ${FUNCTIONS}%"
echo "  Lines:      ${LINES}%"

# 检查是否达标
MIN_COVERAGE=70

if (( $(echo "$STATEMENTS < $MIN_COVERAGE" | bc -l) )); then
  echo "❌ Statement coverage below threshold!"
  send_alert "Low Coverage" "Statements: ${STATEMENTS}%"
  exit 1
fi

if (( $(echo "$BRANCHES < $MIN_COVERAGE" | bc -l) )); then
  echo "❌ Branch coverage below threshold!"
  send_alert "Low Coverage" "Branches: ${BRANCHES}%"
  exit 1
fi

echo "✅ All coverage metrics meet requirements"
exit 0
```

### 3.2 版本健康度监控

**监控维度**:

```typescript
interface VersionHealth {
  version: string;
  releaseDate: Date;
  downloads: number;
  issues: number;
  deprecationStatus: boolean;
  dependenciesResolved: boolean;
}

async function checkVersionHealth(): Promise<VersionHealth[]> {
  const versions = await getAllVersions("@type-dom/svgs");

  return Promise.all(
    versions.map(async (version) => {
      const [downloads, issues] = await Promise.all([
        getDownloads(version),
        getIssuesForVersion(version),
      ]);

      return {
        version,
        downloads,
        issues: issues.length,
        healthScore: calculateHealthScore(downloads, issues.length),
      };
    }),
  );
}

function calculateHealthScore(downloads: number, issues: number): number {
  const downloadScore = Math.min(downloads / 100, 100); // 最多 100 分
  const issuePenalty = issues * 10; // 每个问题扣 10 分

  return Math.max(0, downloadScore - issuePenalty);
}

// 告警规则
if (healthScore < 50) {
  sendCriticalAlert(`Version ${version} health score low: ${healthScore}`);
}
```

  health-check:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: Install dependencies
        run: npm ci

      - name: Run all checks
        run: |
          npm run typecheck
          npm run test
          npm run lint
          npm run build

      - name: Generate health report
        run: node scripts/generate-health-report.js

      - name: Upload report
        uses: actions/upload-artifact@v4
        with:
          name: health-report
          path: reports/health-report.html

      - name: Send daily summary
        if: always()
        uses: slackapi/slack-github-action@v1.24.0
        with:
          payload: |
            {
              "text": "Daily Health Check Summary",
              "blocks": [
                {
                  "type": "section",
                  "text": {
                    "type": "mrkdwn",
                    "text": "*Health Check Results*\n• Type Check: ✅\n• Tests: ✅\n• Build: ✅\n• Coverage: ⚠️ 75%"
                  }
                }
              ]
            }
